20void generuotiFaila(const string& filename, int kiekStudentu, int kiekNd) {
21 ofstream fout(filename);
22 if (!fout) { cerr << "Nepavyko sukurti failo: " << filename << "\n"; return; }
23
24 fout << setw(20) << left << "Pavarde"
25 << setw(20) << left << "Vardas";
26 for (int i = 1; i <= kiekNd; ++i) fout << setw(5) << ("ND" + to_string(i));
27 fout << setw(10) << "Egzaminas" << "\n";
28
29 mt19937 rng(static_cast<unsigned>(time(nullptr)));
30 uniform_int_distribution<int> dist(1, 10);
31
32 for (int i = 1; i <= kiekStudentu; ++i) {
33 string pav = "Pavarde" + to_string(i);
34 string var = "Vardas" + to_string(i);
35
36 fout << setw(20) << left << pav
37 << setw(20) << left << var;
38
39 for (int j = 0; j < kiekNd; ++j) fout << setw(5) << dist(rng);
40 fout << setw(10) << dist(rng) << "\n";
41 }
42
43 cout << "Sugeneruotas failas: " << filename
44 << " (" << kiekStudentu << " irasu)\n";
45}

91void issaugotiIFailaTemplate(const std::string& filename, const Container& students, int metod) {
92 std::ofstream fout(filename, std::ios::out | std::ios::trunc);
93 if (!fout.is_open()) return;
94
95 fout << std::left << std::setw(25) << "Pavarde"
96 << std::setw(25) << "Vardas";
97
98 if (metod == 3) {
99 fout << std::setw(15) << "Galutinis (Vid.)"
100 << std::setw(15) << "Galutinis (Med.)";
101 } else {
102 fout << std::setw(15) << "Galutinis";
103 }
104 fout << "\n";
105
106 std::string buffer;
107 buffer.reserve(1 << 20);
108 const size_t FLUSH_THRESHOLD = (1 << 20);
109
110 for (const auto& s : students) {
111 std::ostringstream line;
112 line << std::left << std::setw(25) << s.pavarde()
113 << std::setw(25) << s.vardas();
114
115 if (metod == 3) {
116 line << std::setw(15) << std::fixed << std::setprecision(2) << s.galVid()
117 << std::setw(15) << std::fixed << std::setprecision(2) << s.galMed();
118 } else {
119 double galutinis = (metod == 1 ? s.galVid() : s.galMed());
120 line << std::setw(15) << std::fixed << std::setprecision(2) << galutinis;
121 }
122 line << "\n";
123
124 buffer.append(line.str());
125 if (buffer.size() > FLUSH_THRESHOLD) {
126 fout.write(buffer.data(), buffer.size());
127 buffer.clear();
128 }
129 }
130
131 if (!buffer.empty()) {
132 fout.write(buffer.data(), buffer.size());
133 }
134}
