19double skaiciuotiMediana(const std::vector<int>& paz) {
20 if (paz.empty()) return 0.0;
21 std::vector<int> tmp = paz;
22 std::sort(tmp.begin(), tmp.end());
23 size_t n = tmp.size();
24 if (n % 2 == 1) return static_cast<double>(tmp[n / 2]);
25 return (static_cast<double>(tmp[n / 2 - 1]) + static_cast<double>(tmp[n / 2])) / 2.0;
26}
