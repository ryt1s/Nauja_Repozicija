54template<typename Container>
55double rikiuoti(Container& studentai, int sortParam, int order) {
56 auto start = high_resolution_clock::now();
57
58 auto comp = [&](const Student& a, const Student& b) {
59 double left = (sortParam == 2 ? a.galMed() : a.galVid());
60 double right = (sortParam == 2 ? b.galMed() : b.galVid());
61 return order == 1 ? left < right : left > right;
62 };
63
64 if constexpr (is_same_v<Container, list<Student>>)
65 studentai.sort(comp);
66 else
67 sort(studentai.begin(), studentai.end(), comp);
68
69 return duration<double>(high_resolution_clock::now() - start).count();
70}

81template<typename Container>
82double splitStudentus(Container& studentai,
83 std::vector<Student>& vargsiukai,
84 std::vector<Student>& kietiakai,
85 SplitStrategy strategy,
86 int sortParam)
87{
88 auto start = high_resolution_clock::now();
89
90 auto isVargsiukas = [&](const Student& s) {
91 return (sortParam == 2 ? s.galMed() : s.galVid()) < 5.0;
92 };
93
94 vargsiukai.clear();
95 kietiakai.clear();
96
97 if (strategy == SplitStrategy::Copy) {
98 for (const auto& s : studentai) {
99 if (isVargsiukas(s)) vargsiukai.push_back(s);
100 else kietiakai.push_back(s);
101 }
102 } else if (strategy == SplitStrategy::Move) {
103 for (auto it = studentai.begin(); it != studentai.end(); ) {
104 if (isVargsiukas(*it)) {
105 vargsiukai.push_back(std::move(*it));
106 it = studentai.erase(it);
107 } else ++it;
108 }
109 kietiakai.assign(studentai.begin(), studentai.end());
110 } else {
111 if constexpr (is_same_v<Container, list<Student>>) {
112 list<Student> good_students;
113 auto it = stable_partition(studentai.begin(), studentai.end(), isVargsiukas);
114 good_students.splice(good_students.begin(), studentai, it, studentai.end());
115 vargsiukai.assign(make_move_iterator(studentai.begin()), make_move_iterator(studentai.end()));
116 kietiakai.assign(make_move_iterator(good_students.begin()), make_move_iterator(good_students.end()));
117 } else {
118 auto it = partition(studentai.begin(), studentai.end(), isVargsiukas);
119 vargsiukai.assign(make_move_iterator(studentai.begin()), make_move_iterator(it));
120 kietiakai.assign(make_move_iterator(it), make_move_iterator(studentai.end()));
121 }
122 }
123
124 return duration<double>(high_resolution_clock::now() - start).count();
125}
